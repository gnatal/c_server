📦 Installation & Umbrella Header

CExpress exposes its entire public API via a single umbrella header:

c 
/* Includes routing, response builders, middlewares, HTTP parser, static files, and JSON */
#include "cexpress.h"

Compile and link directly against the static library libcexpress.a:

bash 
gcc-16 -Wall -Wextra -std=c11 -O2 -Ipath/to/lib -o my_app main.c path/to/libcexpress.a

🚀 Hello World

A minimal server with a single route listening on port 8080:

c 
#include "cexpress.h"
#include <stdio.h>

void handler_home(const Request *req, Response *res) {
    (void)req;
    res_send(res, "Hello from CExpress!");
}

int main(void) {
    App app;
    app_init(&app);

    app_get(&app, "/", handler_home);

    printf("Listening on http://localhost:8080\n");
    app_listen(&app, 8080);

    app_destroy(&app);
    return 0;
}

🧭 Routing & HTTP Verbs

Handlers are defined with the signature void (*Handler)(const Request *req, Response *res). Register handlers for standard HTTP verbs:

c 
app_get(&app, "/users", handler_get_users);
app_post(&app, "/users", handler_create_user);
app_put(&app, "/users/:id", handler_update_user);
app_patch(&app, "/users/:id", handler_patch_user);
app_delete(&app, "/users/:id", handler_delete_user);
Automatic Protocol Conformance:
• HEAD Requests: Executes the matching GET handler while automatically suppressing the message body.
• OPTIONS Requests: Returns 204 No Content with the computed Allow header.
• 405 Method Not Allowed: Emitted automatically if the route path exists but the verb is not registered.

Route Parameters (:id)

Extract URL path variables with req_get_param:

c 
app_get(&app, "/orgs/:org/repos/:repo", handler_repo);

void handler_repo(const Request *req, Response *res) {
    const char *org  = req_get_param(req, "org");
    const char *repo = req_get_param(req, "repo");

    char body[128];
    snprintf(body, sizeof(body), "Repository: %s/%s", org, repo);
    res_send(res, body);
}

Sub-Routers (app_mount)

Modularize routes into isolated Router instances and mount them under path prefixes (like Express's app.use('/api', apiRouter)):

c 
Router api_router;
router_init(&api_router);

/* Scoped to /api */
router_get(&api_router, "/health", handler_health);
router_get(&api_router, "/metrics", handler_metrics);

/* Mounts routes to /api/health and /api/metrics */
app_mount(&app, "/api", &api_router);

🛡️ Middleware Pipeline

Middleware functions match void (*Middleware)(const Request *req, Response *res, MiddlewareChain *chain). Advance with chain_next(chain) or terminate early:

c 
void mw_logger(const Request *req, Response *res, MiddlewareChain *chain) {
    /* 1. Pre-handler processing */
    chain_next(chain);

    /* 2. Post-handler inspection (res->status is finalized) */
    printf("%s %s -> %d\n", req->method, req->path, res->status);
}

/* Register app-wide middleware */
app_use(&app, mw_logger);

/* Register prefix-scoped middleware (only runs for /admin/*) */
app_use_prefix(&app, "/admin", mw_admin_auth);

/* Register per-route middleware */
app_post_mw(&app, "/checkout", handler_checkout, (Middleware[]){mw_auth}, 1);

Centralized Error Handling

Trigger centralized error handling with chain_error:

c 
void mw_guard(const Request *req, Response *res, MiddlewareChain *chain) {
    if (!is_authorized(req)) {
        chain_error(chain, 401, "Invalid or missing API key");
        return;
    }
    chain_next(chain);
}

/* Register custom error handler */
void custom_error_handler(int status, const char *msg, const Request *req, Response *res) {
    (void)req;
    res_status(res, status);
    char json[256];
    snprintf(json, sizeof(json), "{\"error\":\"%s\",\"status\":%d}", msg, status);
    res_json(res, json);
}

app_use_error(&app, custom_error_handler);

📥 Request & Response API

Inspect headers, cookies, query strings, and body data:

c 
void handler_inspect(const Request *req, Response *res) {
    const char *query_sort = req_get_query(req, "sort");
    const char *auth_hdr   = req_get_header(req, "Authorization");
    const char *session    = req_get_cookie(req, "session_id");

    /* Set status and custom headers */
    res_status(res, 200);
    res_set_header(res, "X-Engine", "CExpress");

    /* Send JSON */
    res_json(res, "{\"status\":\"ok\"}");
}

Cookies & Redirects

c 
/* Setting a cookie with SameSite, Max-Age, and HttpOnly */
CookieOptions opts = {
    .max_age = 86400,
    .http_only = 1,
    .secure = 1,
    .same_site = COOKIE_SAMESITE_STRICT
};
res_set_cookie(res, "auth_token", "secure_secret", &opts);

res_clear_cookie(res, "auth_token", "/");

/* Performing a 302 Redirect */
res_redirect(res, 0, "/dashboard");

🌊 Streaming & Chunked Responses

Emit HTTP/1.1 chunked responses incrementally using res_write(), attach RFC 7230 chunked trailers with res_set_trailer(), and finalize with res_end():

c 
void handler_stream(const Request *req, Response *res) {
    (void)req;
    res_set_header(res, "Content-Type", "text/plain");

    /* Stage trailer (declares Trailer: Server-Timing in headers) */
    res_set_trailer(res, "Server-Timing", "app;dur=24.2");

    /* Write chunks incrementally */
    res_write(res, "Chunk 1\n", 8);
    res_write(res, "Chunk 2\n", 8);

    /* Emits terminal 0\r\n, trailers, and terminating CRLF */
    res_end(res);
}

Bounded File Streaming (res_send_file)

Stream large files directly from the filesystem in bounded 16KB chunks through the non-blocking event loop. Memory is bounded and whole files are never buffered into RAM:

c 
void handler_download(const Request *req, Response *res) {
    (void)req;
    if (res_send_file(res, "application/pdf", "reports/report.pdf") != 0) {
        res_status(res, 404);
        res_send(res, "Report not found");
    }
}

Static File Serving (app_serve_static)

Mount a directory of static assets with traversal protection and MIME type detection:

c 
/* GET /static/app.css -> serves public/app.css */
app_serve_static(&app, "/static", "app/public");

📋 Multipart & Form Parsing

Parse multipart/form-data file uploads and application/x-www-form-urlencoded forms:

c 
/* Multipart File Uploads */
const char *content_type = req_get_header(req, "Content-Type");
MultipartForm form;
parse_multipart_body(content_type, req->body, req->content_length, &form);

for (int i = 0; i < form.part_count; i++) {
    if (form.parts[i].is_file) {
        printf("File uploaded: %s (%zu bytes)\n", form.parts[i].filename, form.parts[i].data_len);
    }
}

⚡ Built-in JSON Parser & Serializer

Recursive-descent JSON parser and stringifier without external dependencies:

c 
/* Parse incoming JSON */
char err[128];
JsonValue *val = json_parse(req->body, err, sizeof(err));
if (val != NULL) {
    const JsonValue *name = json_object_get(val, "name");
    json_free(val);
}

/* Construct JSON */
JsonValue *obj = json_new_object();
json_object_set(obj, "status", json_new_string("active"));
char *out = json_stringify(obj);
res_json(res, out);
free(out);
json_free(obj);

🛑 Graceful Shutdown

CExpress intercepts SIGINT and SIGTERM synchronously in the native event loop (EVFILT_SIGNAL on kqueue, signalfd on Linux):

1. Stops accepting incoming TCP connections.
2. Closes idle keep-alive sockets immediately.
3. Sets Connection: close on active responses and streams until flushed.
4. Arms a 5-second deadline timer to prevent stalled clients from blocking process exit.
5. Returns cleanly to main() so app_destroy() frees all memory.

📖 API Cheat Sheet

Function Header Description
app_init(App *app) router.h Initializes server app and dynamic connection table.
app_listen(App *app, int port) connection.h Starts non-blocking event loop (kqueue or epoll).
app_destroy(App *app) connection.h Closes sockets and frees connection tables cleanly.
app_get / app_post / app_put ... router.h Registers route handlers for specific HTTP methods.
app_mount(App *app, prefix, Router *r) router.h Mounts a sub-router under a URL prefix.
app_serve_static(App *app, pfx, root) router.h Mounts static file directory with traversal protection.
app_use(App *app, Middleware mw) router.h Registers an app-wide middleware.
app_use_prefix(App *app, pfx, mw) router.h Registers prefix-scoped middleware (e.g. /api).
chain_next(MiddlewareChain *c) middleware.h Advances to next middleware or route handler.
chain_error(c, status, message) middleware.h Delegates to centralized error handler.
res_status(Response *res, int status) response.h Sets HTTP status code (200, 201, 404, etc.).
res_set_header(res, name, value) response.h Sets a custom response header.
res_send(Response *res, const char *b) response.h Sends text/plain body.
res_json(Response *res, const char *b) response.h Sends application/json body.
res_write(Response *res, data, len) response.h Emits incremental chunk in chunked response.
res_set_trailer(res, name, value) response.h Stages RFC 7230 trailer for chunked transfer.
res_end(Response *res) response.h Finalizes chunked response and emits trailers.
res_send_file(res, type, filepath) response.h Streams file in bounded 16KB chunks through event loop.
req_get_param(req, name) http_parser.h Retrieves extracted route path variable.
req_get_query(req, name) http_parser.h Retrieves parsed query-string parameter.
req_get_header(req, name) http_parser.h Looks up request header (case-insensitive).
